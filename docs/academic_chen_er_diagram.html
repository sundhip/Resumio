Notation Standard: TABLE_NAME (Primary_Key, Attribute_1, Attribute_2, Foreign_Key*)

USERS ( id, email, password_hash, role, status, created_at, updated_at )
/* Core account entity with role domain: ('candidate', 'recruiter', 'admin') */
CANDIDATE_PROFILES ( id, user_id*, full_name, phone, location, headline, bio, photo_url, profile_completion, created_at, updated_at )
/* FK: user_id → USERS(id) [ON DELETE CASCADE], 1:1 relationship with Users */
RECRUITER_PROFILES ( id, user_id*, full_name, phone, company_name, company_logo, industry, location, website, description, profile_completion, created_at, updated_at )
/* FK: user_id → USERS(id) [ON DELETE CASCADE], 1:1 relationship with Users */
CANDIDATE_SKILLS ( id, candidate_profile_id*, name, proficiency, created_at, updated_at )
/* FK: candidate_profile_id → CANDIDATE_PROFILES(id) [ON DELETE CASCADE], Weak Entity Set */
CANDIDATE_EDUCATION ( id, candidate_profile_id*, degree, field_of_study, institution, location, start_date, end_date, currently_studying, grade_or_gpa, description, created_at, updated_at )
/* FK: candidate_profile_id → CANDIDATE_PROFILES(id) [ON DELETE CASCADE], Weak Entity Set */
CANDIDATE_RESUMES ( id, candidate_profile_id*, original_filename, stored_filename, file_type, file_size, is_active, created_at, updated_at )
/* FK: candidate_profile_id → CANDIDATE_PROFILES(id) [ON DELETE CASCADE] */
JOBS ( id, recruiter_id*, recruiter_profile_id*, title, description, responsibilities, company_name, location, employment_type, work_mode, salary_min, salary_max, currency, salary_period, salary_disclosed, min_experience, max_experience, qualification, deadline, status, created_at, updated_at, published_at, closed_at )
/* FK: recruiter_id → USERS(id), recruiter_profile_id → RECRUITER_PROFILES(id) [ON DELETE CASCADE] */
JOB_REQUIRED_SKILLS ( id, job_id*, name, created_at )
/* FK: job_id → JOBS(id) [ON DELETE CASCADE], Decomposed multivalued attribute */
APPLICATIONS ( id, candidate_id*, candidate_profile_id*, job_id*, resume_id*, cover_letter, status, rejection_reason, applied_at, updated_at )
/* Ternary Relationship: candidate_id → USERS(id), job_id → JOBS(id), resume_id → CANDIDATE_RESUMES(id) */
APPLICATION_STATUS_HISTORY ( id, application_id*, old_status, new_status, changed_by*, changed_by_role, note, changed_at )
/* FK: application_id → APPLICATIONS(id) [ON DELETE CASCADE], changed_by → USERS(id) */
INTERVIEWS ( id, application_id*, candidate_id*, recruiter_id*, job_id*, title, interview_type, scheduled_at, duration_minutes, location, meeting_url, description, status, cancellation_reason, reschedule_reason, created_at, updated_at )
/* FK: application_id → APPLICATIONS(id), candidate_id → USERS(id), recruiter_id → USERS(id), job_id → JOBS(id) */
RESUME_JOB_MATCHES ( id, resume_id*, job_id*, application_id*, candidate_id*, candidate_profile_id*, algorithm_version, status, error_message, match_score, required_skills_matched_json, required_skills_missing_json, preferred_skills_matched_json, preferred_skills_missing_json, required_skills_score, preferred_skills_score, experience_score, qualification_score, relevance_score, started_at, completed_at, created_at, updated_at )
/* AI deterministic score evaluation matching snapshot resume against job requirements */
NOTIFICATIONS ( id, user_id*, type, title, message, related_entity_type, related_entity_id, is_read, created_at )
/* FK: user_id → USERS(id) [ON DELETE CASCADE] */
RESUME_PARSED_DATA ( id, resume_id*, candidate_id*, candidate_profile_id*, file_hash, parser_version, status, raw_text, candidate_name, email, phone, location, headline, summary, skills_json, education_json, experience_json, projects_json, certifications_json, started_at, completed_at, created_at, updated_at )
/* 1:1 structured extraction of uploaded resume document */
